# M2 · 视觉完型整合 `gestalt` Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现第五个游戏「视觉完型整合」——短暂暴露一张碎片化的目标轮廓图，掩蔽后从 2×2 的相似图形阵列中指出它，并把它作为已注册的 `GameModule` 接入自适应核心。

**Architecture:** 图形库是**数据表 + 生成器**（`figures.ts` 里 27 个图形各是一段声明式描述，`buildFigure` 统一展开成折线），碎片化是纯函数（`fragment.ts`，弧长切段 + 保留率抽样 + 噪声 + 旋转），试次构造是纯函数（`rules.ts`，档位表 + 同族/异族干扰项 + 2×2 选项阵列），`engine.ts` 是只认 `start(now)`/`tick(now)`/`input(e.t)` 帧时间戳的阶段状态机，`view.ts` 只做 SVG 节点增删与事件转发、**完全不持有时间**，`index.ts` 组装成 `GameModule`（`judge` / `summarize` / `formatSummary` / `staircase` / `instructions`）。

**Tech Stack:** TypeScript 5.9.3 `strict` + `verbatimModuleSyntax`、Vite 8.3.4、Vitest 5.0.5（**node 环境，无 jsdom、无 canvas**）、fake-indexeddb 6.2.6。**本里程碑不新增任何依赖。**

**Spec:** `docs/superpowers/specs/2026-10-18-focus-trainer-design.md`（本游戏：§3.5 参数表 144–178 行、§4.2 接口 219–227 行、§5.2 升降档 311 行、§5.3 能力指数 317 行、§6 帧间隔 361 行、§7 完型碎片测试要求 373 行）

## Global Constraints

以下每条对本计划所有 Task 隐含生效，字面值直接抄自 spec 与 M1 已落地的代码契约：

- **不新增依赖**。不 `npm install`，不引 jsdom / canvas / 图形库。视图只用 `document.createElementNS` + SVG。
- **TypeScript 5.9.3 `strict` + `verbatimModuleSyntax`**：任何只用于类型的位置必须 `import type`；不允许 `any` 逃逸，不允许非空断言滥用。
- **时钟契约（spec §4.3 / §6）**：轮内计时**只**用 rAF 帧时间戳（`engine.start(now)` / `engine.tick(now)`）与 `PointerEvent.timeStamp`（视图 emit 时透传）。`Date.now()` 只允许用于 seed、记录时间戳、日键。**本游戏 `rules.ts` / `fragment.ts` / `figures.ts` / `engine.ts` / `view.ts` 五个文件里出现 `setTimeout`、`setInterval`、`Date.now()`、`requestAnimationFrame` 任一字面量即为返工触发条件**（`view.ts` 尤其：视图不做计时）。

```ts
export const CURVE_SAMPLES = 96;

const rad = (deg: number): number => (deg * Math.PI) / 180;

function regularPts(n: number, rotDeg: number): Pt[] {
  return Array.from({ length: n }, (_, i) => {
    const a = rad(rotDeg) + (2 * Math.PI * i) / n;
    return [Math.cos(a), Math.sin(a)] as Pt;
  });
}

function starPts(n: number, inner: number, rotDeg: number): Pt[] {
  return Array.from({ length: 2 * n }, (_, i) => {
    const a = rad(rotDeg) + (Math.PI * i) / n;
    const r = i % 2 === 0 ? 1 : inner;
    return [r * Math.cos(a), r * Math.sin(a)] as Pt;
  });
}

/** 按弧长等距重采样。closed 时把「末点→首点」那一跳也算进总长。 */
function resample(raw: readonly Pt[], count: number, closed: boolean): Pt[] {
  const loop = closed ? [...raw, raw[0]] : [...raw];
  const cum: number[] = [0];
  for (let i = 1; i < loop.length; i++) {
    cum.push(cum[i - 1] + Math.hypot(loop[i][0] - loop[i - 1][0], loop[i][1] - loop[i - 1][1]));
  }
  const total = cum[cum.length - 1];
  const out: Pt[] = [];
  let seg = 1;
  for (let j = 0; j < count; j++) {
    const target = closed ? (total * j) / count : (total * j) / (count - 1);
    while (seg < cum.length - 1 && cum[seg] < target) seg += 1;
    const span = cum[seg] - cum[seg - 1] || 1;
    const f = (target - cum[seg - 1]) / span;
    out.push([
      loop[seg - 1][0] + (loop[seg][0] - loop[seg - 1][0]) * f,
      loop[seg - 1][1] + (loop[seg][1] - loop[seg - 1][1]) * f,
    ]);
  }
  return out;
}

function paramPts(f: (t: number) => Pt, closed: boolean): Pt[] {
  return Array.from({ length: DENSE }, (_, i) => f((2 * Math.PI * i) / (DENSE - (closed ? 0 : 1))));
}

/** 等比缩放到 [-1,1]²：取所有点的最大绝对坐标做分母，保长宽比。 */
function normalize(strokes: readonly FigureStroke[]): readonly FigureStroke[] {
  let m = 0;
  for (const s of strokes) for (const [x, y] of s.pts) m = Math.max(m, Math.abs(x), Math.abs(y));
  if (m === 0 || m === 1) return strokes.map((s) => ({ pts: s.pts.map((p) => [p[0], p[1]] as Pt), closed: s.closed }));
  return strokes.map((s) => ({
    pts: s.pts.map((p) => [p[0] / m, p[1] / m] as Pt),
    closed: s.closed,
  }));
}

function expand(shape: Shape): readonly FigureStroke[] {
  switch (shape.kind) {
    case 'regular':
      return [{ pts: regularPts(shape.n, shape.rotDeg), closed: true }];
    case 'star':
      return [{ pts: starPts(shape.n, shape.inner, shape.rotDeg), closed: true }];
    case 'ellipse':
      return [{
        pts: resample(paramPts((t) => [shape.rx * Math.cos(t), shape.ry * Math.sin(t)], true), CURVE_SAMPLES, true),
        closed: true,
      }];
    case 'named':
      return [{
        pts: resample(
          shape.which === 'heart'
            ? paramPts((t) => [
                16 * Math.pow(Math.sin(t), 3),
                13 * Math.cos(t) - 5 * Math.cos(2 * t) - 2 * Math.cos(3 * t) - Math.cos(4 * t),
              ], true)
            : paramPts((t) => [Math.cos(t), Math.sin(t) * Math.pow(Math.sin(t / 2), 2)], true),
          CURVE_SAMPLES,
          true,
        ),
        closed: true,
      }];
    case 'crescent': {
      const a0 = rad(38);
      const a1 = rad(322);
      const outer = paramPtsOn((i) => i / (DENSE - 1), a0, a1, (a) => [Math.cos(a), Math.sin(a)]);
      const innerRaw: Pt[] = [];
      for (let i = 0; i < DENSE; i++) {
        const a = a1 + ((a0 - a1) * i) / (DENSE - 1);
        const x = shape.cx + shape.ri * Math.cos(a);
        const y = shape.ri * Math.sin(a);
        if (x * x + y * y <= 1.0000001) innerRaw.push([x, y]);
      }
      if (innerRaw.length < 2) throw new Error('crescent 内弧退化了');
      return [{ pts: resample([...outer, ...innerRaw], CURVE_SAMPLES, true), closed: true }];
    }
    case 'strokes':
      return shape.strokes.map((s) => ({ pts: s.pts.map((p) => [p[0], p[1]] as Pt), closed: s.closed }));
  }
}

function paramPtsOn(
  frac: (i: number) => number,
  a0: number,
  a1: number,
  f: (a: number) => Pt,
): Pt[] {
  return Array.from({ length: DENSE }, (_, i) => f(a0 + (a1 - a0) * frac(i)));
}

const CACHE = new Map<FigureId, FigureGeometry>();

export function buildFigure(id: FigureId): FigureGeometry {
  const hit = CACHE.get(id);
  if (hit) return hit;
  const spec = FIGURE_INDEX.get(id);
  if (spec === undefined) throw new RangeError(`未知的完型图形 id: ${id}`);
  const geom: FigureGeometry = { id, family: spec.family, strokes: normalize(expand(spec.shape)) };
  CACHE.set(id, geom);
  return geom;
}
```

```ts
import { createRng, randInt, shuffle } from '../../core/rng';
import type { Rng } from '../../core/rng';
import { FAMILIES, FIGURES } from './figures';
import type { Family, FigureId, FigureSpec } from './figures';

export type RotationMode = 'none' | 'jitter30' | 'full360';
export type DistractorRule = 'cross' | 'same';

export interface GestaltLevel {
  readonly exposureMs: number;
  readonly retention: number;
  readonly noise: number;
  readonly rotation: RotationMode;
  readonly distractor: DistractorRule;
  /** 「相似」的字面量化：本档允许的同族干扰项数量上限。cross → 0，same → 3。 */
  readonly sameFamily: 0 | 3;
}

export const GESTALT_CHOICE_COUNT = 4;

/** spec §3.5 档位表（12 档），逐字面抄，不做公式化插值。 */
export const GESTALT_LEVELS: readonly GestaltLevel[] = [
  { exposureMs: 600, retention: 0.75, noise: 0, rotation: 'none', distractor: 'cross', sameFamily: 0 },
  { exposureMs: 500, retention: 0.7, noise: 0, rotation: 'none', distractor: 'cross', sameFamily: 0 },
  { exposureMs: 420, retention: 0.66, noise: 5, rotation: 'none', distractor: 'cross', sameFamily: 0 },
  { exposureMs: 350, retention: 0.62, noise: 10, rotation: 'none', distractor: 'cross', sameFamily: 0 },
  { exposureMs: 300, retention: 0.58, noise: 15, rotation: 'jitter30', distractor: 'cross', sameFamily: 0 },
  { exposureMs: 250, retention: 0.54, noise: 20, rotation: 'jitter30', distractor: 'cross', sameFamily: 0 },
  { exposureMs: 210, retention: 0.5, noise: 25, rotation: 'jitter30', distractor: 'same', sameFamily: 3 },
  { exposureMs: 180, retention: 0.47, noise: 30, rotation: 'jitter30', distractor: 'same', sameFamily: 3 },
  { exposureMs: 150, retention: 0.44, noise: 35, rotation: 'full360', distractor: 'same', sameFamily: 3 },
  { exposureMs: 125, retention: 0.41, noise: 40, rotation: 'full360', distractor: 'same', sameFamily: 3 },
  { exposureMs: 100, retention: 0.38, noise: 50, rotation: 'full360', distractor: 'same', sameFamily: 3 },
  { exposureMs: 80, retention: 0.35, noise: 60, rotation: 'full360', distractor: 'same', sameFamily: 3 },
];

export const GESTALT_MAX_LEVEL = GESTALT_LEVELS.length;

export function gestaltLevel(level: number): GestaltLevel {
  if (!Number.isInteger(level) || level < 1 || level > GESTALT_MAX_LEVEL) {
    throw new RangeError(`gestalt level out of range: ${level}`);
  }
  return GESTALT_LEVELS[level - 1];
}

export function rotationDeg(rng: Rng, mode: RotationMode): number {
  if (mode === 'none') return 0;
  if (mode === 'jitter30') return -30 + rng() * 60;
  return rng() * 360;
}

export interface GestaltTrialSpec {
  readonly target: FigureId;
  readonly targetFamily: Family;
  readonly choices: readonly FigureId[];
  readonly correctIndex: number;
  readonly sameFamilyDistractors: number;
  readonly crossFamilyDistractors: number;
}

export function pickTarget(rng: Rng, pool: readonly FigureSpec[] = FIGURES): FigureSpec {
  if (pool.length === 0) throw new RangeError('完型图形池为空');
  return shuffle(rng, pool)[0];
}

/**
 * 产出最终 4 个选项（顺序已洗牌，含目标本身）。
 * spec:157 ——「异族」抽 3 个别的族；「同族」从本族抽，不够 3 个时从其他族补足。
 * pool 参数是为了让「族太小」这条边界可用假池子测，不改动 FIGURES。
 */
export function buildChoiceSpecs(
  rng: Rng,
  target: FigureSpec,
  lv: GestaltLevel,
  pool: readonly FigureSpec[] = FIGURES,
): FigureSpec[] {
  const want = GESTALT_CHOICE_COUNT - 1;
  const samePool = pool.filter((f) => f.family === target.family && f.id !== target.id);
  const crossPool = pool.filter((f) => f.family !== target.family);
  if (1 + samePool.length + crossPool.length < GESTALT_CHOICE_COUNT) {
    throw new RangeError(`图形池只有 ${1 + samePool.length + crossPool.length} 个可用图形，凑不出 ${GESTALT_CHOICE_COUNT} 个互不相同的选项`);
  }
  const sameTake = shuffle(rng, samePool).slice(0, Math.min(lv.sameFamily, samePool.length, want));
  const crossTake = shuffle(rng, crossPool).slice(0, want - sameTake.length);
  return shuffle(rng, [target, ...sameTake, ...crossTake]);
}

export function buildChoices(
  rng: Rng,
  target: FigureSpec,
  lv: GestaltLevel,
  pool: readonly FigureSpec[] = FIGURES,
): FigureId[] {
  return buildChoiceSpecs(rng, target, lv, pool).map((f) => f.id);
}

export function buildTrialSpec(
  rng: Rng,
  lv: GestaltLevel,
  pool: readonly FigureSpec[] = FIGURES,
): GestaltTrialSpec {
  const target = pickTarget(rng, pool);
  const specs = buildChoiceSpecs(rng, target, lv, pool);
  const same = specs.filter((f) => f.family === target.family && f.id !== target.id).length;
  return {
    target: target.id,
    targetFamily: target.family,
    choices: specs.map((f) => f.id),
    correctIndex: specs.findIndex((f) => f.id === target.id),
    sameFamilyDistractors: same,
    crossFamilyDistractors: specs.length - 1 - same,
  };
}

// FAMILIES / randInt 由本模块的测试与后续 Task 的采样用得上；此处显式引用以免被 tree-shake
// 成「未使用」而删掉再造成 diff 抖动。
export const GESTALT_FAMILIES: readonly Family[] = FAMILIES;
export const _randInt = randInt;
```

最后一行 `_randInt` 是**权宜导出**：如果执行者发现 `randInt` / `createRng` 在本文件确实没用到，就**直接从 import 里删掉**（不要留 `_` 前缀的死导出）；`FAMILIES` 同理，若无人用 `GESTALT_FAMILIES` 就删这一行，保持 `import` 只列真正用到的符号。`strict` 下未使用变量不会报错，但计划要求不留死码。

**裁定 —— 切段的三处 spec 没说死的细节**：
1. **零头**：一条轮廓除以 0.08 一般有余数。规则 = 余段长度 ≥ `SEG_LEN / 2` 时保留为一段（否则丢弃），保证碎片长度分布窄，人眼仍读到连续轮廓。
2. **旋转作用于「整体」**（spec:155「……均匀撒在 [-1,1]² 内；整体旋转 θ」）：先按 `rng` 依次抽保留段、再抽 `n` 段噪声，最后对**碎片与噪声一起**乘同一个 θ。噪声段中心取 `[-1+SEG_LEN/2, 1-SEG_LEN/2]`，保证整段落在 `[-1,1]²` 内（否则高档位会把线段画到 viewBox 外面）。
3. `retention` 是**独立伯努利**（spec:154「每段独立以保留率 k 保留」），不是「抽固定比例」；每段恰好消耗一次 `rng()`，所以同种子逐段一致。

```ts
import type { Rng } from '../../core/rng';
import type { FigureGeometry, Pt } from './figures';

/** spec:155 —— 沿轮廓按弧长切成长 0.08 的小段。 */
export const SEG_LEN = 0.08;
/** 掩蔽帧铺满屏的随机线段数（spec 只说「满屏随机线段」，数量在这里定死）。 */
export const MASK_SEGMENTS = 48;

export interface Segment {
  readonly a: Pt;
  readonly b: Pt;
}

export interface FragmentOptions {
  readonly retention: number;
  readonly noise: number;
  readonly rotationDeg: number;
}

export interface FragmentedFigure {
  readonly segments: readonly Segment[];
  readonly noise: readonly Segment[];
  readonly totalSegs: number;
  readonly keptSegs: number;
  readonly actualRetention: number;
  readonly rotationDeg: number;
}

const dist = (p: Pt, q: Pt): number => Math.hypot(q[0] - p[0], q[1] - p[1]);

/**
 * 逐笔切段：闭合笔把「末点→首点」那一跳也算进去。
 * 一条笔的弧长一般不是 SEG_LEN 的整数倍，零头只在 ≥ SEG_LEN/2 时保留为最后一段。
 */
export function cutSegments(geom: FigureGeometry, segLen: number = SEG_LEN): Segment[] {
  const out: Segment[] = [];
  for (const st of geom.strokes) {
    const loop = st.closed ? [...st.pts, st.pts[0]] : [...st.pts];
    let from: Pt = loop[0];
    let carry = 0;
    for (let i = 0; i < loop.length - 1; i++) {
      const a = loop[i];
      const b = loop[i + 1];
      let cursor = a;
      let rest = dist(a, b);
      while (rest > 1e-12) {
        const need = segLen - carry;
        if (rest >= need) {
          const t = need / rest;
          const to: Pt = [cursor[0] + (b[0] - cursor[0]) * t, cursor[1] + (b[1] - cursor[1]) * t];
          out.push({ a: from, b: to });
          carry = 0;
          cursor = to;
          from = to;
          rest -= need;
        } else {
          carry += rest;
          rest = 0;
          if (carry >= segLen / 2) {
            out.push({ a: from, b });
            carry = 0;
            from = b;
          }
        }
      }
    }
  }
  return out;
}

export function rotate(segs: readonly Segment[], deg: number): Segment[] {
  const r = (deg * Math.PI) / 180;
  const rawC = Math.cos(r);
  const rawS = Math.sin(r);
  // 90/180/270 度把 6e-17 量级的尾巴抹成精确 0，否则等值断言与像素对齐都会飘
  const c = Math.abs(rawC) < 1e-15 ? 0 : rawC;
  const s = Math.abs(rawS) < 1e-15 ? 0 : rawS;
  const rot = (p: Pt): Pt => [p[0] * c - p[1] * s, p[0] * s + p[1] * c];
  return segs.map((sg) => ({ a: rot(sg.a), b: rot(sg.b) }));
}

function noiseSegments(rng: Rng, n: number): Segment[] {
  const half = SEG_LEN / 2;
  const lim = 1 - half;
  const out: Segment[] = [];
  for (let i = 0; i < n; i++) {
    const cx = -lim + rng() * 2 * lim;
    const cy = -lim + rng() * 2 * lim;
    const angle = rng() * Math.PI * 2;
    const dx = half * Math.cos(angle);
    const dy = half * Math.sin(angle);
    out.push({ a: [cx - dx, cy - dy], b: [cx + dx, cy + dy] });
  }
  return out;
}

export function fragmentFigure(geom: FigureGeometry, rng: Rng, opt: FragmentOptions): FragmentedFigure {
  const all = cutSegments(geom);
  const kept: Segment[] = [];
  for (const seg of all) if (rng() < opt.retention) kept.push(seg);
  const noise = noiseSegments(rng, opt.noise);
  return {
    segments: rotate(kept, opt.rotationDeg),
    noise: rotate(noise, opt.rotationDeg),
    totalSegs: all.length,
    keptSegs: kept.length,
    actualRetention: all.length === 0 ? 0 : kept.length / all.length,
    rotationDeg: opt.rotationDeg,
  };
}

/** 掩蔽帧：与目标无关的满屏随机线段。 */
export function makeMask(rng: Rng, n: number = MASK_SEGMENTS): Segment[] {
  return noiseSegments(rng, n);
}
```

`rotate` 恒返回**新建对象**，任何一次调用都不把上游数组的对象引用透出 —— 这是「视图对快照只读」这条约束在数据层的落点。

- Produces:

  - `type GestaltPhase = 'idle' | 'fixation' | 'exposure' | 'mask' | 'answer' | 'feedback' | 'done'`
  - `interface GestaltChoiceView { readonly id: FigureId; readonly strokes: readonly FigureStroke[] }`
  - `interface GestaltSnapshot { phase; elapsedMs; fragments; mask; choices; answeredIndex; correctIndex; rtMs; exposureMs; timingValid; done }`
  - `class GestaltEngine implements GameEngine<GestaltSnapshot>`

**裁定（本 Task 的四条硬性设计，不留给执行者判断）**：

1. **曝光时长按帧边界量化**。`configure` 只算出「请求值」`lv.exposureMs`；实际起止完全由 `start(now)` / `tick(now)` 的帧时间戳决定：进入暴露期的那一帧是 `exposureStartAt`，之后**第一帧 `now >= exposureStartAt + T`** 结束。所以 `实测 exposureMs ∈ [T, T + 帧长)`，而这个实测值才是 `识别阈值` 的输入（spec:178「实测暴露时长」）。
2. **一个 tick 最多推进一个阶段**（不做 `while` 追赶）。故意的：一次巨大卡顿如果一口气跨过暴露+掩蔽，用户会根本没见过图形却被要求作答；一帧一阶段 + 阶段内的帧间隔判定会把这轮判成 `timingValid = false`，交给 `SessionRunner` 补轮。
3. **帧间隔只在暴露期与掩蔽期检查**（spec:6 明文把「完型的暴露与掩蔽」列为时序关键阶段；舒尔特恒 `true` 那条不照抄）。判定规则：`tick` 进入时**若上一帧结束时所处阶段是 `exposure` 或 `mask`**，则 `now - lastFrameAt > 50` 记一次违规。**进入暴露期的那一帧**自己的间隔不算（它属于注视期），离开掩蔽期的那一帧同理。`lastFrameAt` 是显式字段，不是钳制。
4. **拷贝口径**：`snapshot()` 对三个线段数组做 `slice()`（数组级防御拷贝），但**共享 `Segment` / `FigureStroke` 对象**——理由：这些对象只在 `configure` 里由 `fragmentFigure` / `buildFigure` 新建，之后引擎与视图都没有任何写路径（`readonly` + 无 setter），逐帧深拷贝 60–135 个对象是白扔 60fps 的预算。`result()` 对 `trials` 逐条 `{ ...t }` + 新建 `metrics` 对象；gestalt 的 trial 全是原始值字段，所以这一层浅拷贝在该深度上就是完整防御拷贝。

```ts
import { createRng } from '../../core/rng';
import type { GameEngine, InputEvent, RoundResult, TrialRecord } from '../../core/types';
import { buildFigure } from './figures';
import type { FigureId, FigureStroke } from './figures';
import { MASK_SEGMENTS, fragmentFigure, makeMask } from './fragment';
import type { Segment } from './fragment';
import { GESTALT_CHOICE_COUNT, buildTrialSpec, gestaltLevel, rotationDeg } from './rules';
import type { GestaltLevel, GestaltTrialSpec } from './rules';

export const FIXATION_MS = 500;
export const MASK_MS = 200;
export const FEEDBACK_MS = 600;
/** spec §6：时序关键阶段内任一帧间隔 > 50ms → timingValid = false。 */
export const FRAME_BUDGET_MS = 50;

export type GestaltPhase = 'idle' | 'fixation' | 'exposure' | 'mask' | 'answer' | 'feedback' | 'done';

export interface GestaltChoiceView {
  readonly id: FigureId;
  readonly strokes: readonly FigureStroke[];
}

export interface GestaltSnapshot {
  readonly phase: GestaltPhase;
  readonly elapsedMs: number;
  readonly fragments: readonly Segment[];
  readonly mask: readonly Segment[];
  readonly choices: readonly GestaltChoiceView[];
  readonly answeredIndex: number | null;
  readonly correctIndex: number | null;
  readonly rtMs: number;
  readonly exposureMs: number;
  readonly timingValid: boolean;
  readonly done: boolean;
}

export class GestaltEngine implements GameEngine<GestaltSnapshot> {
  private level = 1;
  private seed = 0;
  private roundNo = 0;
  private lv: GestaltLevel = gestaltLevel(1);
  private trial: GestaltTrialSpec | null = null;
  private fragments: readonly Segment[] = [];
  private maskSegs: readonly Segment[] = [];
  private choiceViews: readonly GestaltChoiceView[] = [];

  private phase: GestaltPhase = 'idle';
  private phaseStartAt = 0;
  private phaseUntil = 0;
  private roundStartAt = 0;
  /** 显式「上一帧时间戳」字段：单调性靠它，不靠钳制。 */
  private lastFrameAt = 0;
  private exposureStartAt = 0;
  private exposureMs = 0;
  private answeredIndex: number | null = null;
  private answeredAt = 0;
  private correctIndex: number | null = null;
  private rt = 0;
  private timingValid = true;
  private now = 0;
  private endAt: number | null = null;
  private trials: TrialRecord[] = [];

  configure(level: number, seed: number, roundNo: number): void {
    // 先校验档位，越界时不改任何状态（configure 抛错不该留下半轮）
    const lv = gestaltLevel(level);
    const rng = createRng(seed);
    const trial = buildTrialSpec(rng, lv);
    const rotation = rotationDeg(rng, lv.rotation);
    const frag = fragmentFigure(buildFigure(trial.target), rng, {
      retention: lv.retention,
      noise: lv.noise,
      rotationDeg: rotation,
    });
    const mask = makeMask(rng, MASK_SEGMENTS);
    const choices = trial.choices.map((id) => ({ id, strokes: buildFigure(id).strokes }));

    this.level = level;
    this.seed = seed;
    this.roundNo = roundNo;
    this.lv = lv;
    this.trial = trial;
    this.fragments = frag.segments;
    this.maskSegs = mask;
    this.choiceViews = choices;
    // 作废契约（spec 5.1）：SessionRunner.voidCurrentRound() 不通知引擎，重开只调 configure + start。
    // 下面每一行都是「清上一轮残留」，少一行就会漏出作废轮的作答、时序判定或阶段。
    this.phase = 'idle';
    this.phaseStartAt = 0;
    this.phaseUntil = 0;
    this.roundStartAt = 0;
    this.lastFrameAt = 0;
    this.exposureStartAt = 0;
    this.exposureMs = 0;
    this.answeredIndex = null;
    this.answeredAt = 0;
    this.correctIndex = null;
    this.rt = 0;
    this.timingValid = true;
    this.now = 0;
    this.endAt = null;
    this.trials = [];
  }

  start(now: number): void {
    this.phase = 'fixation';
    this.phaseStartAt = now;
    this.phaseUntil = now + FIXATION_MS;
    this.roundStartAt = now;
    this.lastFrameAt = now;
    this.now = now;
  }

  input(e: InputEvent): void {
    if (e.kind !== 'choice') return;
    this.now = Math.max(this.now, e.t);
    if (this.phase !== 'answer' || this.answeredIndex !== null || this.trial === null) return;
    if (e.index < 0 || e.index >= GESTALT_CHOICE_COUNT) return;
    const correct = e.index === this.trial.correctIndex;
    this.answeredIndex = e.index;
    this.answeredAt = e.t;
    this.rt = e.t - this.exposureStartAt;
    this.correctIndex = this.trial.correctIndex; // 反馈期才揭晓
    this.trials = [
      {
        onset: this.exposureStartAt,
        rt: this.rt,
        correct,
        target: this.trial.target,
        targetFamily: this.trial.targetFamily,
        choice: e.index,
        sameFamilyDistractors: this.trial.sameFamilyDistractors,
        exposureMs: this.exposureMs,
      },
    ];
    this.phase = 'feedback';
    this.phaseStartAt = e.t;
    this.phaseUntil = e.t + FEEDBACK_MS;
  }

  tick(now: number): void {
    this.now = Math.max(this.now, now);
    if (this.phase === 'idle' || this.phase === 'done') return;
    // 帧间隔只在「上一帧仍处于时序关键阶段」时判定
    if (this.phase === 'exposure' || this.phase === 'mask') {
      if (now - this.lastFrameAt > FRAME_BUDGET_MS) this.timingValid = false;
    }
    this.lastFrameAt = now;
    if (now < this.phaseUntil) return;
    // 一个 tick 最多推进一个阶段
    if (this.phase === 'fixation') {
      this.phase = 'exposure';
      this.exposureStartAt = now;
      this.phaseUntil = now + this.lv.exposureMs;
    } else if (this.phase === 'exposure') {
      this.exposureMs = now - this.exposureStartAt;
      this.phase = 'mask';
      this.phaseUntil = now + MASK_MS;
    } else if (this.phase === 'mask') {
      this.phase = 'answer';
      this.phaseUntil = Number.POSITIVE_INFINITY; // 作答期不设上限
    } else if (this.phase === 'feedback') {
      this.phase = 'done';
      this.endAt = now;
    }
    this.phaseStartAt = now;
  }

  snapshot(): GestaltSnapshot {
    const inExposure = this.phase === 'exposure';
    const inMask = this.phase === 'mask';
    const showChoices = this.phase === 'answer' || this.phase === 'feedback' || this.phase === 'done';
    const end = this.endAt ?? this.now;
    return {
      phase: this.phase,
      elapsedMs: this.roundStartAt === 0 ? 0 : Math.max(0, end - this.roundStartAt),
      fragments: inExposure ? this.fragments.slice() : [],
      mask: inMask || (this.phase === 'answer' && false) ? this.maskSegs.slice() : inMask ? this.maskSegs.slice() : [],
      choices: showChoices ? this.choiceViews.slice() : [],
      answeredIndex: this.answeredIndex,
      correctIndex: this.phase === 'feedback' || this.phase === 'done' ? this.correctIndex : null,
      rtMs: this.rt,
      exposureMs: this.exposureMs,
      timingValid: this.timingValid,
      done: this.endAt !== null,
    };
  }

  isDone(): boolean {
    return this.endAt !== null;
  }

  result(): RoundResult {
    if (this.endAt === null) throw new Error('gestalt round is not finished');
    return {
      level: this.level,
      seed: this.seed,
      trials: this.trials.map((t) => ({ ...t })),
      metrics: {
        correct: this.trials.length === 1 && this.trials[0].correct ? 1 : 0,
        rtMs: Math.round(this.rt),
        exposureMs: this.exposureMs,
        requestedExposureMs: this.lv.exposureMs,
      },
      timingValid: this.timingValid,
    };
  }
}
```

**裁定**：
- **只用 DOM/SVG 节点，不用 `<canvas>`**。理由有两条硬的：① Vitest 跑在 node 环境，没有 jsdom 也没有 canvas 2D 上下文，`getContext('2d')` 在这套测试里根本无法被断言；② SVG 的 `<polyline points="…">` 是纯字符串属性，手写 DOM 桩能直接核对「暴露期到底渲染了几段碎片、答案阵列有没有提前出现」。SVG 的几何变换全部由引擎在 `configure` 里算完（旋转、归一化），视图**不做数学**，只做坐标→字符串。
- `http://www.w3.org/2000/svg` 是本项目唯一允许出现的 URL 字面量（M4 的 APK 审计只放行 `w3.org`），除此之外任何外部地址都不许出现。
- **视图零计时**：`view.ts` 里不允许出现 `setTimeout` / `setInterval` / `Date.now()` / `requestAnimationFrame`。阶段与剩余时间全部从快照读，倒计时由引擎的 `tick` 驱动。Step 1 的最后一条测试把这条钉成可执行断言（扫源码字面量）。
- 事件映射：`{ kind: 'choice', index, t: ev.timeStamp }`，`index` 是 2×2 阵列的 0..3（阅读序：左上、右上、左下、右下）。`t` 必须取 `PointerEvent.timeStamp`，与 rAF 同域（`session.ts` 把它喂给 `engine.input` 与 `completeRound`）。

```ts
import type { GameView, InputEvent } from '../../core/types';
import type { GestaltChoiceView, GestaltSnapshot } from './engine';
import type { FigureStroke, Pt } from './figures';
import type { Segment } from './fragment';

/** 本项目唯一允许的 URL 字面量：M4 的 APK 审计只放行 w3.org。 */
export const SVG_NS = 'http://www.w3.org/2000/svg';

const VIEW_BOX = '-1.15 -1.15 2.3 2.3';

const f = (n: number): string => n.toFixed(3);

/** 图形坐标 y 轴向上、SVG 向下，这里统一取反；闭合笔补回首点。 */
export function pointsAttr(pts: readonly Pt[], closed: boolean): string {
  const loop = closed ? [...pts, pts[0]] : pts;
  return loop.map((p) => `${f(p[0])},${f(-p[1])}`).join(' ');
}

export function segmentPoints(s: Segment): string {
  return `${f(s.a[0])},${f(-s.a[1])} ${f(s.b[0])},${f(-s.b[1])}`;
}

function svg(children: SVGElement[]): SVGElement {
  const el = document.createElementNS(SVG_NS, 'svg') as unknown as SVGElement;
  el.setAttribute('viewBox', VIEW_BOX);
  el.setAttribute('class', 'gestalt-svg');
  el.append(...(children as unknown as Node[]));
  return el;
}

function polyline(points: string): SVGElement {
  const el = document.createElementNS(SVG_NS, 'polyline') as unknown as SVGElement;
  el.setAttribute('points', points);
  el.setAttribute('class', 'gestalt-seg');
  return el;
}

function figureSvg(c: GestaltChoiceView): SVGElement {
  return svg(c.strokes.map((s: FigureStroke) => polyline(pointsAttr(s.pts, s.closed))));
}

export function createGestaltView(host: HTMLElement, emit: (e: InputEvent) => void): GameView<GestaltSnapshot> {
  const root = document.createElement('div');
  root.className = 'gestalt';
  const info = document.createElement('div');
  info.className = 'gestalt-info';
  const stage = document.createElement('div');
  stage.className = 'gestalt-stage';
  root.append(info, stage);
  host.append(root);

  let layoutKey = '';
  let buttons: HTMLElement[] = [];

  const onDown = (ev: PointerEvent): void => {
    const hit = ev.target instanceof Element ? ev.target.closest('button[data-i]') : null;
    if (!(hit instanceof HTMLButtonElement)) return;
    ev.preventDefault();
    emit({ kind: 'choice', index: Number(hit.dataset.i), t: ev.timeStamp });
  };
  stage.addEventListener('pointerdown', onDown);

  const build = (s: GestaltSnapshot): void => {
    if (s.choices.length > 0) {
      buttons = s.choices.map((c, i) => {
        const b = document.createElement('button') as unknown as HTMLElement;
        b.className = 'gestalt-choice';
        b.dataset.i = String(i);
        b.append(figureSvg(c) as unknown as Node);
        return b;
      });
      stage.replaceChildren(...(buttons as unknown as Node[]));
      return;
    }
    buttons = [];
    if (s.fragments.length > 0) {
      stage.replaceChildren(svg(s.fragments.map((g) => polyline(segmentPoints(g)))) as unknown as Node);
    } else if (s.mask.length > 0) {
      stage.replaceChildren(svg(s.mask.map((g) => polyline(segmentPoints(g)))) as unknown as Node);
    } else {
      const mark = document.createElement('div');
      mark.className = 'gestalt-fixation';
      mark.textContent = '+';
      stage.replaceChildren(mark);
    }
  };

  return {
    render(s) {
      // 键含阶段 + 内容长度 + 选项 id：阶段一变就整体重建，杜绝「暴露期碎片残留进作答期」。
      const key = `${s.phase}|${s.fragments.length}|${s.mask.length}|${s.choices.map((c) => c.id).join(',')}`;
      if (key !== layoutKey) {
        layoutKey = key;
        build(s);
      }
      if (s.choices.length > 0 && (s.phase === 'feedback' || s.phase === 'done')) {
        buttons.forEach((b, i) => {
          b.classList.toggle('right', s.correctIndex === i);
          b.classList.toggle('wrong', s.answeredIndex === i && s.correctIndex !== i);
        });
      }
      info.textContent =
        s.phase === 'answer'
          ? '选出刚才看到的图形'
          : s.phase === 'feedback' || s.phase === 'done'
            ? `反应 ${(s.rtMs / 1000).toFixed(2)} 秒`
            : ' ';
    },
    destroy() {
      stage.removeEventListener('pointerdown', onDown);
      root.remove();
    },
  };
}
```

`build()` 在作答期只按选项 id 序列重建节点，反馈期改的是 class 而不是重建，避免闪烁；`svg()` / `polyline()` 上的 `as unknown as SVGElement` 是因为本项目不引入 `@types/*` 之外的 DOM 层封装，M1 的 `view.ts` 同样直接吃全局 `document`。若执行者能把 `SVGElement` 换成 `StubEl` 兼容的更干净写法（例如让辅助函数返回 `Element`），可以，但**不得新增依赖**。

```ts
/** spec:176-177 —— 选对 pass 得 1，选错 fail 得 0；没有 hold，也不掺反应时。 */
export function judgeGestalt(r: RoundResult): { outcome: Outcome; score01: number } {
  const correct = r.metrics.correct === 1;
  return { outcome: correct ? 'pass' : 'fail', score01: correct ? 1 : 0 };
}

/** spec:178 —— 识别阈值取最后 6 个计分试次的实测暴露时长均值。 */
export const THRESHOLD_WINDOW = 6;

export function summarizeGestalt(rounds: readonly RoundRecord[]): Record<string, number> {
  const counted = rounds.filter((r) => r.counted);
  if (counted.length === 0) return { accuracy: 0, thresholdMs: 0 };
  const hits = counted.filter((r) => r.metrics.correct === 1).length;
  const tail = counted.slice(Math.max(0, counted.length - THRESHOLD_WINDOW));
  const mean = tail.reduce((sum, r) => sum + r.metrics.exposureMs, 0) / tail.length;
  return { accuracy: Math.round((hits / counted.length) * 1000) / 1000, thresholdMs: Math.round(mean) };
}

export function formatGestaltSummary(s: Record<string, number>): SummaryItem[] {
  return [
    { label: '正确率', value: `${Math.round(s.accuracy * 100)}%` },
    { label: '识别阈值', value: `${s.thresholdMs} 毫秒` },
  ];
}
```
